        // Draw node markers
        visibleNodes.forEach((node, index) => {
            const pos = nodePositions[index];
            const nodeGroup = g.append('g').attr('transform', `translate(${spineX}, ${pos.top})`);

            // Node circle
            nodeGroup
                .append('circle')

                .attr('fill', nodeTypeColors[node.type] || '#6b7280')
